cmake_minimum_required(VERSION 3.22.1)
project(spine_vulkan LANGUAGES CXX)
find_library(log-lib log)
find_library(android-lib android)
find_library(vulkan-lib vulkan)
add_library(spine_vulkan SHARED spine_vulkan.cpp)
target_compile_features(spine_vulkan PRIVATE cxx_std_17)
target_link_libraries(spine_vulkan PRIVATE ${vulkan-lib} ${android-lib} ${log-lib})
